# 如何利用随机数求签

``` python
import re
import random


def cleanline(line):
    return line.strip().replace("\u3000", "")


def getCount(line):
    line = cleanline(line)
    a = re.search("第([0-9]+?)簽", line)
    if a:
        return int(a.group(1))
    else:
        return -1

def getdata():
    with open("./黄大仙签文.txt", encoding = 'utf8') as f:
        lines = f.readlines()
        result = dict()
        a = ""
        n = 0
        for line in lines:
            b = getCount(line)
            if b > 0:
                if a:
                    result[n] = a
                n = b
                a = ""
            a += cleanline(line) + '\n'
        result[n] = a
    return result

def getRandom(n, comment):
    result = getdata()
    a = int(random.random() * 100)
    b = n + a
    b = (b-1) % 100 + 1
    string = result[b]
    print("要问的问题是：")
    print(comment)
    print("输入的随机数是：{}， 电脑随机数是：{}，最终结果是：".format(n, a))
    print(string)


if __name__ == "__main__":
    n = 18
    comment = "xx问，今年股票会不会涨？"
    getRandom(n, comment)
```
